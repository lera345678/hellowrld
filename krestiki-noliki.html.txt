# создана функция, которая будет отображать текущее состояние игрового поля в консоли
board = list(range(1, 10))
# функция для отображения игрового поля.
def draw_board(board):
    print("-" * 13)
    # проходим по строкам поля и выводим каждый ряд:
    for i in range(3):
        print(board[i * 3:(i + 1) * 3])  # получаем каждый ряд поля по 3 элемента:
# для ввода пользователем
def take_input(player_token):
    valid = False
    while not valid:
        player_answer = input("Куда ставим " + player_token + "? ")
        try:
            player_answer = int(player_answer)  # для преобразования строки - int()
        except:
            # если введено, что-то кроме чисел от 1 до 9
            print("Некорректный ввод. Вы уверены, что ввели число?")
            continue
        if player_answer >= 1 and player_answer <= 9:
            if (str(board[player_answer - 1]) not in "XO"):
                board[player_answer - 1] = player_token
                valid = True
            else:
                print("Здесь уже занято")
        else:
            print("Введите число от 1 до 9")
# проверка игрового поля, создаем кортеж
def check_win(board):
    # создаем все возможные выигрышные комбинации:
    win_coord = ((0, 1, 2), (3, 4, 5), (6, 7, 8), (0, 3, 6), (1, 4, 7), (2, 5, 8), (0, 4, 8), (2, 4, 6))
    for each in win_coord:
        if board[each[0]] == board[each[1]] == board[
            each[2]]:  # проверяет совпадают ли значения в трех клетках комбинаций
            return board[each[0]]
    return False
def main(board):
    counter = 0  # счетчик ходов.
    win = False
    while not win:  # создаем цикл while(выполняется, пока никто из игроков не выиграл)
        draw_board(board)
        if counter % 2 == 0:  # определение хода
            take_input("X")
        else:
            take_input("O")
        counter += 1  # увеличиваем счетчик ходов.
        if counter > 4:  # проверка победителя с 5 хода
            tmp = check_win(board)  # проверка есть ли победитель
            if tmp:
                print(tmp, "выиграл!")
                win = True
                break
        if counter == 9:
            print("Ничья!")
            break  # завершение игры
    draw_board(board)  # конечное состояние поля


main(board)
